rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper function to check if user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // Helper function to check if user owns the document
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }
    
    // Users collection
    match /users/{userId} {
      // Allow users to read their own profile
      allow get: if isAuthenticated() && isOwner(userId);
      
      // Allow users to read public profiles (for search/suggestions)
      allow get: if isAuthenticated() && resource.data.publicProfile == true;
      
      // Allow querying usernames for availability check (needed during signup)
      // This allows checking if username exists before creating account
      allow list: if true; // Allow anyone to query (needed for username check)
      
      // Allow users to write only their own profile
      allow create: if isAuthenticated() && isOwner(userId);
      allow update: if isAuthenticated() && isOwner(userId);
    }
    
    // Sessions collection
    match /sessions/{sessionId} {
      // Allow reading own sessions
      allow read: if isAuthenticated() && 
        resource.data.userId == request.auth.uid;
      
      // Allow reading friend sessions (for leaderboard)
      // Note: This checks if the session owner is in current user's friends list
      allow read: if isAuthenticated() && 
        resource.data.userId != request.auth.uid &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.friends != null &&
        resource.data.userId in get(/databases/$(database)/documents/users/$(request.auth.uid)).data.friends;
      
      // Allow creating sessions with correct userId
      allow create: if isAuthenticated() && 
        request.resource.data.userId == request.auth.uid;
      
      // Allow updating own sessions
      allow update: if isAuthenticated() && 
        resource.data.userId == request.auth.uid;
      
      // Allow deleting own sessions
      allow delete: if isAuthenticated() && 
        resource.data.userId == request.auth.uid;
    }
    
    // Friend requests collection
    match /friendRequests/{requestId} {
      // Allow reading own friend requests (sent or received)
      allow get: if isAuthenticated() && (
        resource.data.toUserId == request.auth.uid || 
        resource.data.fromUserId == request.auth.uid
      );
      
      // Allow listing friend requests - queries filter by toUserId/fromUserId
      allow list: if isAuthenticated();
      
      // Allow creating friend requests
      allow create: if isAuthenticated() && 
        request.resource.data.fromUserId == request.auth.uid;
      
      // Allow updating friend requests (only recipient can update)
      allow update: if isAuthenticated() && 
        resource.data.toUserId == request.auth.uid;
    }
  }
}
